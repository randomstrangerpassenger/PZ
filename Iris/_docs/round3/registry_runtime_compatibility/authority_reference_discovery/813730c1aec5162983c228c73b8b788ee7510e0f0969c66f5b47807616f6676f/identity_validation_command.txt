function Get-Sha256([byte[]]$bytes) {
    $sha = [Security.Cryptography.SHA256]::Create()
    try {
        [BitConverter]::ToString($sha.ComputeHash($bytes)).Replace('-', '').ToLowerInvariant()
    }
    finally {
        $sha.Dispose()
    }
}
function Get-LengthFrame([uint64]$length) {
    $bytes = [BitConverter]::GetBytes($length)
    [Array]::Reverse($bytes)
    $bytes
}
$identityFields = @(
    '23cb60a1354eed6bcc8d9b9735e8c7b388cd1fdcb75726b93bc299ef22dd9334',
    '7400c85f2267880a83e7b98dff250460151bc022839bde39e304d876c7dfe703',
    '["-S","-c"]',
    'C',
    'C',
    'UTC',
    'UTF-8'
)
$identityStream = New-Object IO.MemoryStream
foreach ($field in $identityFields) {
    $fieldBytes = [Text.Encoding]::UTF8.GetBytes($field)
    $lengthBytes = Get-LengthFrame $fieldBytes.Length
    $identityStream.Write($lengthBytes, 0, $lengthBytes.Length)
    $identityStream.Write($fieldBytes, 0, $fieldBytes.Length)
}
$canonicalizerIdentity = $identityStream.ToArray()
function Get-DiscoveryId([string]$json) {
    $canonicalLine = $json | jq -S -c .
    if ($LASTEXITCODE -ne 0) {
        throw 'jq canonicalization failed'
    }
    $canonicalBytes = [Text.Encoding]::UTF8.GetBytes($canonicalLine + "`r")
    $preimage = New-Object IO.MemoryStream
    $prefix = [Text.Encoding]::UTF8.GetBytes("iris-rtc-discovery-id-v1`n")
    $preimage.Write($prefix, 0, $prefix.Length)
    $lengthBytes = Get-LengthFrame $canonicalizerIdentity.Length
    $preimage.Write($lengthBytes, 0, $lengthBytes.Length)
    $preimage.Write($canonicalizerIdentity, 0, $canonicalizerIdentity.Length)
    $lengthBytes = Get-LengthFrame $canonicalBytes.Length
    $preimage.Write($lengthBytes, 0, $lengthBytes.Length)
    $preimage.Write($canonicalBytes, 0, $canonicalBytes.Length)
    Get-Sha256 $preimage.ToArray()
}

$valuePaths = @(jq --stream -r 'select(length==2)|.[0]|@json' -- $core)
$duplicateKeyCount = @($valuePaths | Group-Object | Where-Object { $_.Count -gt 1 }).Count
$syntheticValuePaths = @('{"a":1,"a":2}' | jq --stream -r 'select(length==2)|.[0]|@json')
$syntheticDuplicateKeyCount = @($syntheticValuePaths | Group-Object | Where-Object { $_.Count -gt 1 }).Count
